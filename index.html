import yfinance as yf import pandas as pd import numpy as np # ========================== # DAFTAR SAHAM IDX # ========================== TICKERS = [ "BBCA.JK","BBRI.JK","BMRI.JK","BBNI.JK", "TLKM.JK","ASII.JK","ICBP.JK","INDF.JK", "ANTM.JK","MDKA.JK","TINS.JK","BRPT.JK", "TPIA.JK","IMPC.JK","CPIN.JK","JPFA.JK", "GOTO.JK","AMMN.JK","PGEO.JK","AKRA.JK", "EXCL.JK","ISAT.JK","MAPI.JK","UNTR.JK", "ADRO.JK","ITMG.JK","PTBA.JK","ESSA.JK", "MEDC.JK","HRUM.JK" ] results = [] for ticker in TICKERS: try: df = yf.download( ticker, period="1y", interval="1d", progress=False, auto_adjust=True ) if len(df) < 220: continue close = df["Close"] volume = df["Volume"] ma20 = close.rolling(20).mean() ma50 = close.rolling(50).mean() ma200 = close.rolling(200).mean() current_price = float(close.iloc[-1]) current_ma20 = float(ma20.iloc[-1]) current_ma50 = float(ma50.iloc[-1]) current_ma200 = float(ma200.iloc[-1]) current_volume = float(volume.iloc[-1]) avg_volume20 = float(volume.tail(20).mean()) distance_ma200 = ( abs(current_price - current_ma200) / current_ma200 ) * 100 # OBV obv = (np.sign(close.diff()) * volume).fillna(0).cumsum() obv_now = float(obv.iloc[-1]) obv_20 = float(obv.iloc[-20]) obv_up = obv_now > obv_20 volume_ratio = current_volume / avg_volume20 score = 0 if current_price > current_ma20: score += 20 if current_price > current_ma50: score += 20 if volume_ratio > 2: score += 30 if obv_up: score += 20 if distance_ma200 <= 10: score += 10 if ( current_price > current_ma20 and current_price > current_ma50 and volume_ratio > 1.5 and distance_ma200 <= 10 ): results.append({ "Ticker": ticker.replace(".JK",""), "Price": round(current_price,2), "MA200": round(current_ma200,2), "Distance MA200 %": round(distance_ma200,2), "Volume Ratio": round(volume_ratio,2), "OBV Up": obv_up, "Accumulation Score": score }) except: pass result_df = pd.DataFrame(results) if not result_df.empty: result_df = result_df.sort_values( by="Accumulation Score", ascending=False ) print("\nTOP SAHAM AKUMULASI BANDAR\n") print(result_df.head(10).to_string(index=False)) else: print("Tidak ada saham yang memenuhi kriteria.")
